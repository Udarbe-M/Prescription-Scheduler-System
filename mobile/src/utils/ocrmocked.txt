// Parse extracted text to find medication information
export const parsePrescriptionText = (text: string): ExtractedData => {
  const extracted: ExtractedData = {};
  const lowerText = text.toLowerCase();
  
  // Look for dosage pattern (e.g., 500mg, 10mg, 850mg)
  const dosageRegex = /(\d+(?:\.\d+)?)\s*(mg|g|ml|mcg|µg|tablet|tab|cap|capsule|pill)\b/i;
  const dosageMatch = text.match(dosageRegex);
  if (dosageMatch) {
    extracted.dosage = dosageMatch[0];
  }
  
  // Look for frequency with times per day
  const frequencyRegex = /(\d+)\s*times?\s*(?:a|per|)\s*(day|week)\b/i;
  const freqMatch = lowerText.match(frequencyRegex);
  
  if (freqMatch) {
    const times = parseInt(freqMatch[1]);
    const period = freqMatch[2];
    
    if (period === 'day') {
      if (times === 1) extracted.frequency = 'daily';
      else if (times === 2) extracted.frequency = 'twice';
      else if (times === 3) extracted.frequency = 'thrice';
    } else if (period === 'week') {
      extracted.frequency = 'weekly';
    }
  } else {
    // Fallback to keyword matching
    if (lowerText.includes('three times') || lowerText.includes('thrice') || lowerText.includes('3 times')) {
      extracted.frequency = 'thrice';
    } else if (lowerText.includes('twice') || lowerText.includes('two times') || lowerText.includes('2 times')) {
      extracted.frequency = 'twice';
    } else if (lowerText.includes('once') || lowerText.includes('daily') || lowerText.includes('1 time')) {
      extracted.frequency = 'daily';
    } else if (lowerText.includes('weekly') || lowerText.includes('week')) {
      extracted.frequency = 'weekly';
    }
  }
  
  // Common medication names (expand this list)
  const commonMeds = [
    'amoxicillin', 'ibuprofen', 'paracetamol', 'aspirin', 'metformin',
    'atorvastatin', 'lisinopril', 'levothyroxine', 'metoprolol', 'prednisone',
    'omeprazole', 'sertraline', 'simvastatin', 'amlodipine', 'hydrochlorothiazide'
  ];
  
  // Try to find medication name
  for (const med of commonMeds) {
    if (lowerText.includes(med)) {
      extracted.name = med.charAt(0).toUpperCase() + med.slice(1);
      break;
    }
  }
  
  // If not found in common list, look for capitalized words that might be medication names
  if (!extracted.name) {
    const words = text.split(/\s+/);
    for (const word of words) {
      if (word.length > 3 && /^[A-Z][a-z]+$/.test(word)) {
        // Check if it's not a common non-medication word
        const commonNonMeds = ['Patient', 'Doctor', 'Clinic', 'Hospital', 'Prescription', 'Rx', 'Sig'];
        if (!commonNonMeds.includes(word)) {
          extracted.name = word;
          break;
        }
      }
    }
  }
  
  // Extract instructions from common phrases
  const instructionKeywords = [
    { keyword: 'with food', instruction: 'Take with food' },
    { keyword: 'after eating', instruction: 'Take after meals' },
    { keyword: 'before meal', instruction: 'Take before meals' },
    { keyword: 'after meal', instruction: 'Take after meals' },
    { keyword: 'empty stomach', instruction: 'Take on empty stomach' },
    { keyword: 'bedtime', instruction: 'Take at bedtime' },
    { keyword: 'before bed', instruction: 'Take before bed' },
    { keyword: 'morning', instruction: 'Take in the morning' },
    { keyword: 'evening', instruction: 'Take in the evening' },
    { keyword: 'as needed', instruction: 'Take as needed' },
    { keyword: 'with water', instruction: 'Take with water' },
  ];
  
  for (const { keyword, instruction } of instructionKeywords) {
    if (lowerText.includes(keyword) && !extracted.instructions) {
      extracted.instructions = instruction;
    }
  }
  
  return extracted;
};
